# 0. Поля Справочник.Сотрудники (подтвердить org-поле: Организация / ГоловнаяОрганизация)

орг_поле = "ГоловнаяОрганизация" if "ГоловнаяОрганизация" in поля else (
    "Организация" if "Организация" in поля else None)
print("Сотрудники.org-поле:", орг_поле, "| есть ФизическоеЛицо:", "ФизическоеЛицо" in поля)

        pass

        if рс.Пустой():
            статус = "Нет сотрудника"
        else:
            # ПодрОрг (реверс по А_ПодразделениеЗарплатаБух)
            стрп = орг.А_ПодразделениеЗарплатаБух if EZ(орг) else None

            if EZ(орг) and EZ(стрп):
                qp = erp.NewObject("Запрос")
                qp.УстановитьПараметр("Орг", орг); qp.УстановитьПараметр("СП", стрп)
                qp.Текст = ("ВЫБРАТЬ ПЕРВЫЕ 1 ПО.Ссылка КАК Ссылка ИЗ Справочник.ПодразделенияОрганизаций КАК ПО "
                            "ГДЕ ПО.Организация = &Орг И ПО.А_Подразделение = &СП И НЕ ПО.ПометкаУдаления")
                рп = qp.Выполнить()
                if not рп.Пустой():
                    подр = "ok"
            if подр is None:
                статус = "Нет подразделения"
    if статус == "OK":
        matched_sum[едр] += сумма
    else:
        unmatched.append((едр, имя, drfo, статус, round(сумма, 2)))

# 4. Отчёт + ассерты
print("\n=== Σ per орг (BuhBud Кт vs сопоставлено) ===")
for едр in sorted(buh_sum):
    print(f"  {едр}: BuhBud={buh_sum[едр]:.2f} | matched={matched_sum[едр]:.2f} | "
          f"diff={buh_sum[едр]-matched_sum[едр]:.2f}")
print(f"\nНесопоставленных строк: {len(unmatched)} (Σ={sum(u[4] for u in unmatched):.2f})")
for u in unmatched[:40]:
    print("  ", u)

assert sum(buh_sum.values()) > 0, "BuhBud вернул 0 — нет остатков 661/663"
