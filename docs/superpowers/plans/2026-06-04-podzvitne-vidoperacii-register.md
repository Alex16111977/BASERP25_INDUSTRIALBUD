```python
# -*- coding: utf-8 -*-
import io, sys
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
import win32com.client as w
erp = w.Dispatch("V83.COMConnector").Connect('Srvr="localhost";Ref="BaseERP";Usr="Администратор";Pwd="24043"')
енум = erp.Перечисления.ВидыОперацийПоЗарплате.УдержаниеНеизрасходованныхПодотчетныхСумм

for i in range(ob.УдержанияБухгалтерия.Количество()):
    s = ob.УдержанияБухгалтерия.Получить(i)
    if erp.ЗначениеЗаполнено(s.ФизическоеЛицо) and s.ФизическоеЛицо.Наименование.startswith("Гуняга"):
        вид = s.ВидОперации.Наименование if erp.ЗначениеЗаполнено(s.ВидОперации) else "<нет>"
        ст = s.СтатьяАктивовПассивов.Наименование if erp.ЗначениеЗаполнено(s.СтатьяАктивовПассивов) else "<нет>"
        сч = s.СчетУчета.Код if erp.ЗначениеЗаполнено(s.СчетУчета) else "<нет>"
        суб = s.Субконто1.Наименование if erp.ЗначениеЗаполнено(s.Субконто1) else "<нет>"
        print(f"Гуняга: ВидОперации={вид} | статья={ст} | счёт={сч} | Субконто1={суб}")
```

```python
# -*- coding: utf-8 -*-
import io, sys
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
import win32com.client as w
erp = w.Dispatch("V83.COMConnector").Connect('Srvr="localhost";Ref="BaseERP";Usr="Администратор";Pwd="24043"')
ob = erp.Документы.А_ОтражениеЗПпоКазне.НайтиПоНомеру("000000001").ПолучитьОбъект()
ob.ЗагрузитьНалогиИУдержанияИзЗарплатыБух()
ok = False
for i in range(ob.УдержанияБухгалтерия.Количество()):
    s = ob.УдержанияБухгалтерия.Получить(i)
    флн = s.ФизическоеЛицо.Наименование if erp.ЗначениеЗаполнено(s.ФизическоеЛицо) else ""
    вид = s.ВидОперации.Наименование if erp.ЗначениеЗаполнено(s.ВидОперации) else "<нет>"
    ст = s.СтатьяАктивовПассивов.Наименование if erp.ЗначениеЗаполнено(s.СтатьяАктивовПассивов) else "<пусто>"
    сч = s.СчетУчета.Код if erp.ЗначениеЗаполнено(s.СчетУчета) else "<нет>"
    суб = s.Субконто1.Наименование if erp.ЗначениеЗаполнено(s.Субконто1) else "<нет>"
    print(f"   {флн[:22]:22s} | вид={вид[:30]:30s} | счёт={сч} | статья={ст[:20]:20s} | Субконто1={суб}")
    if флн.startswith("Гуняга"):
        ok = (вид.startswith("Удержание неизрасход") and сч == "3721" and ст == "<пусто>"
              and суб == флн)
print("ИТОГ Гуняга:", "GREEN ✅" if ok else "RED ❌")
```
