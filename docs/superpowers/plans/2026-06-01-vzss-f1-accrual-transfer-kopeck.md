**Goal:** Устранить копеечный разрыв (−0.01) в `РегистрНакопления.А_ВзаиморасчетыССотрудниками`: по Форме1 начисления (NET «Зарплата управл») и их налоги (НДФЛ/ВС) должны совпадать до 0.01 между дочерним `ОтражениеЗарплатыВФинансовомУчете` (Приход) и `А_ПередачаНачисленийМеждуПодразделениями` (перенос).

**Architecture:** Корень — официальный Ф1-НДФЛ в `А_ОтражениеЗПпоКазне` разбит по подразделениям ДВАЖДЫ по разным ключам: gross-up строки «Удержание с админ зарплату» в `НачисленнаяЗарплатаИВзносыПоФизлицам` (Op2, равномерно) и ТЧ `НачисленныйНДФЛ` (Op3, налог-за-налогом). Фикс — пост-проход, перестраивающий gross-up «Удержание» Ф1 под `НачисленныйНДФЛ` per (ФЛ, подразделение), сохраняя Σ per ФЛ. Разрыв «начислено↔выплачено» (110.26/128.58) — норма, НЕ трогаем.

// per (ФізЛицо, Подразделение). Один і той самий офіційний НДФЛ розбивається по
// підрозділах у двох місцях (Op2 «Удержание» рівномірно vs Op3 НачисленныйНДФЛ

// рядки під НачисленныйНДФЛ. Σ per ФЛ зберігається (НДФЛДляФЛ = Σ НачисленныйНДФЛ).
// Викликається ПІСЛЯ _ОтрЗП_РазметитьФормаPL (коли ФормаPL вже проставлено обом ТЧ).

	Ф1 = Перечисления.А_ФормыPL.Форма1;

	// 1) НачисленныйНДФЛ Ф1 per (ФЛ, Подр) — зберігаємо посилання
	НДФЛКлюч = Новый Соответствие;
	Для Каждого Стр Из НачисленныйНДФЛ Цикл
		Если Стр.ФормаPL <> Ф1 Тогда Продолжить; КонецЕсли;
		К = XMLСтрока(Стр.ФизическоеЛицо) + "|" + XMLСтрока(Стр.ПодразделениеПредприятия);
		Зап = НДФЛКлюч[К];
		Если Зап = Неопределено Тогда
			Зап = Новый Структура("ФЛ, Подр, Сумма", Стр.ФизическоеЛицо, Стр.ПодразделениеПредприятия, 0);
			НДФЛКлюч.Вставить(К, Зап);
		КонецЕсли;
		Зап.Сумма = Зап.Сумма + Стр.Сумма;
	КонецЦикла;

	// 2) Зібрати рядки «Удержание с админ» Ф1: шаблон per ФЛ + індекси на видалення
	Шаблон = Новый Соответствие;
	НаУдаление = Новый Массив;
	Для і = 0 По НачисленнаяЗарплатаИВзносыПоФизлицам.Количество() - 1 Цикл
		Стр = НачисленнаяЗарплатаИВзносыПоФизлицам[і];
		Если Стр.ФормаPL <> Ф1 Тогда Продолжить; КонецЕсли;
		Способ = Стр.СпособОтраженияЗарплатыВБухучете;
		Если НЕ (ЗначениеЗаполнено(Способ) И Способ.А_ЭтоУдержание) Тогда Продолжить; КонецЕсли;
		Если Шаблон[Стр.ФизическоеЛицо] = Неопределено Тогда
			Т = Новый Структура("Способ, ОргБух, ВидОперации, ГруппаУчета");
			Т.Способ      = Стр.СпособОтраженияЗарплатыВБухучете;
			Т.ОргБух      = Стр.ОрганизацияБухгалтерия;
			Т.ВидОперации = Стр.ВидОперации;
			Т.ГруппаУчета = Стр.ГруппаУчетаНачислений;
			Шаблон.Вставить(Стр.ФизическоеЛицо, Т);
		КонецЕсли;
		НаУдаление.Добавить(і);
	КонецЦикла;

	// 3) Видалити старі «Удержание» рядки (з кінця — індекси не зсуваються)
	і = НаУдаление.Количество() - 1;
	Пока і >= 0 Цикл
		НачисленнаяЗарплатаИВзносыПоФизлицам.Удалить(НаУдаление[і]);
		 і = і - 1;
	КонецЦикла;

	// 4) Додати по (ФЛ, Подр) = НачисленныйНДФЛ
	Для Каждого КЗ Из НДФЛКлюч Цикл
		З = КЗ.Значение;
		Если З.Сумма = 0 Тогда Продолжить; КонецЕсли;
		Т = Шаблон[З.ФЛ];
		Если Т = Неопределено Тогда Продолжить; КонецЕсли; // у ФЛ не було gross-up «Удержание» — пропускаємо
		НовСтр = НачисленнаяЗарплатаИВзносыПоФизлицам.Добавить();
		НовСтр.Организация                       = Организация;
		НовСтр.ФизическоеЛицо                    = З.ФЛ;
		НовСтр.ПодразделениеПредприятия          = З.Подр;
		НовСтр.ВидОперации                       = Т.ВидОперации;
		НовСтр.ГруппаУчетаНачислений             = Т.ГруппаУчета;
		НовСтр.СпособОтраженияЗарплатыВБухучете  = Т.Способ;
		НовСтр.Сумма                             = З.Сумма;
		НовСтр.ВзносыВсего                       = 0;
		НовСтр.ФормаPL                           = Ф1;
		Если ЗначениеЗаполнено(Т.ОргБух) Тогда
			НовСтр.ОрганизацияБухгалтерия = Т.ОргБух;
		КонецЕсли;
	КонецЦикла;

КонецПроцедуры
