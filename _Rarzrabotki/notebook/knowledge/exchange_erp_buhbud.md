	Если НЕ ЗначениеЗаполнено(Строка.Партнер)
		ИЛИ ОбщегоНазначения.ЗначенияРеквизитовОбъекта(Объект.Ссылка, "Контрагент").Контрагент <> Объект.Контрагент Тогда

		Если НЕ ЗначениеЗаполнено(Строка.Партнер) Тогда

Для Каждого СтрокаРасшифровки Из Объект.РасшифровкаПлатежа Цикл
	Если Объект.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ОплатаПоставщику
		ИЛИ Объект.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ВозвратОплатыКлиенту Тогда
		
		
		ПараметрыОР = ОбъектыРасчетовСервер.ПолучитьПараметрыОбъектаРасчетов();
		ПараметрыОР.Организация = Объект.Организация;
		ПараметрыОР.Партнер     = СтрокаРасшифровки.Партнер;
		ПараметрыОР.Контрагент  = Объект.Контрагент;
		ПараметрыОР.Вставить("ПометкаУдаления",Ложь) ;
		
		Источник = Неопределено;
		ТипРасчетов = Неопределено;
		ТипОР = Неопределено;
		Договор = СтрокаРасшифровки.А_Договор;
		
		Если Объект.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ОплатаПоставщику Тогда
			ТипРасчетов = Перечисления.ТипыРасчетовСПартнерами.РасчетыСПоставщиком;
		ИначеЕсли Объект.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ВозвратОплатыКлиенту Тогда
			ТипРасчетов = Перечисления.ТипыРасчетовСПартнерами.РасчетыСКлиентом;
		КонецЕсли;
		
		Если ЗначениеЗаполнено(Договор) Тогда
			ПорядокРасчетов = Договор.ПорядокРасчетов;
		Иначе
			ПорядокРасчетов = Перечисления.ПорядокРасчетов.ПустаяСсылка();
		КонецЕсли;
		
		Если ПорядокРасчетов = Перечисления.ПорядокРасчетов.ПоЗаказамНакладным Тогда 
			
			Если  ТипРасчетов = Перечисления.ТипыРасчетовСПартнерами.РасчетыСКлиентом 
				И ЗначениеЗаполнено(СтрокаРасшифровки.А_ЗаказКлиента) Тогда
				Источник = СтрокаРасшифровки.А_ЗаказКлиента;   
				ТипОР = Перечисления.ТипыОбъектовРасчетов.Заказ; 
				Объект.Договор = Договор;
                Объект.ГруппаФинансовогоУчета = Договор.ГруппаФинансовогоУчета;
			ИначеЕсли  ТипРасчетов = Перечисления.ТипыРасчетовСПартнерами.РасчетыСПоставщиком 
				И ЗначениеЗаполнено(СтрокаРасшифровки.А_ЗаказПоставщику) Тогда
				Источник = СтрокаРасшифровки.А_ЗаказПоставщику; 
				ТипОР = Перечисления.ТипыОбъектовРасчетов.Заказ;
				Объект.Договор = Договор;
				Объект.ГруппаФинансовогоУчета = Договор.ГруппаФинансовогоУчета;
			Иначе
				Источник = Договор;
				ТипОР = Перечисления.ТипыОбъектовРасчетов.Договор;
			КонецЕсли;
			
		ИначеЕсли ЗначениеЗаполнено(Договор) Тогда
			Источник = Договор;
			ТипОР = Перечисления.ТипыОбъектовРасчетов.Договор;
		Иначе
			Продолжить;
		КонецЕсли;
		
		ПараметрыОР.Объект = Источник;
		ПараметрыОР.ТипРасчетов = ТипРасчетов;
		ПараметрыОР.Договор = Договор;
